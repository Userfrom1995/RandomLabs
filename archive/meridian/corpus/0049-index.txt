<script>
(function () {
  'use strict';
  var H = window.Halcyon;
  var source = document.getElementById('source');
  var stdin = document.getElementById('stdin');
  var select = document.getElementById('example-select');

  var exampleNames = Object.keys(H.examples).sort();
  exampleNames.forEach(function (name) {
    var opt = document.createElement('option');
    opt.value = name;
    opt.textContent = name;
    select.appendChild(opt);
  });
  select.value = 'fib.hly';
  source.value = H.examples['fib.hly'];

  select.addEventListener('change', function () {
    source.value = H.examples[select.value] || '';
  });

  function stdinLines() {
    var t = stdin.value;
    if (t === '') { return []; }
    return t.split('\n');
  }

  // Effect-aware CLI output: effect output plus the rendered final value
  // (nothing when it is the unit value), mirroring the Haskell CLI.
  function cliText(res, show) {
    if (!res || res.kind) { return ''; }
    var v = res.value;
    var unitK = v && (v.k === 'unit' || v.k === 'vm_unit');
    return (res.out || '') + (unitK ? '' : show(v));
  }

  // -------------------------------------------------------------------
  // Tabs
  // -------------------------------------------------------------------
  var tabs = document.querySelectorAll('.tab');
  tabs.forEach(function (tab) {
    tab.addEventListener('click', function () {
      tabs.forEach(function (t) { t.classList.remove('active'); });
      document.querySelectorAll('.panel').forEach(function (p) { p.classList.remove('active'); });
      tab.classList.add('active');
      document.getElementById('panel-' + tab.getAttribute('data-panel')).classList.add('active');
    });
  });

  function show(panel, text, isError) {
    var el = document.getElementById('out-' + panel);
    el.textContent = text;
    el.className = isError ? 'error' : '';
  }
  function showError(panel, err) {
    var pos = err && err.pos && err.pos.line > 0
      ? 'line ' + err.pos.line + ', col ' + err.pos.col + ': '
      : '';
    show(panel, 'error (' + err.kind + '): ' + pos + err.message, true);
  }

  function activate(panel) {
    tabs.forEach(function (t) { t.classList.toggle('active', t.getAttribute('data-panel') === panel); });
    document.querySelectorAll('.panel').forEach(function (p) { p.classList.toggle('active', p.id === 'panel-' + panel); });
  }

  // -------------------------------------------------------------------
  // AST rendering
  // -------------------------------------------------------------------
  function renderAst(e, depth) {
    var pad = new Array(depth * 2 + 1).join(' ');
    function pos(e) { return e.pos ? '@' + e.pos.line + ':' + e.pos.col : ''; }
    switch (e.kind) {
      case 'int':    return pad + 'int ' + e.v + ' ' + pos(e);
      case 'float':  return pad + 'float ' + e.v + ' ' + pos(e);
      case 'bool':   return pad + 'bool ' + e.v + ' ' + pos(e);
      case 'str':    return pad + 'str "' + e.v + '" ' + pos(e);
      case 'char':   return pad + 'char ' + e.v + ' ' + pos(e);
      case 'unit':   return pad + 'unit ' + pos(e);
      case 'var':    return pad + 'var ' + e.name + ' ' + pos(e);
      case 'builtin': return pad + 'builtin ' + e.name + ' ' + pos(e);
      case 'constr': return pad + 'constr ' + e.name + ' ' + pos(e);
      case 'method': return pad + 'method ' + e.name + ' ' + pos(e);
      case 'list': {
        var lines = [pad + 'list ' + pos(e)];
        e.items.forEach(function (x) { lines.push(renderAst(x, depth + 1)); });
        return lines.join('\n');
      }
      case 'lambda': {
        var l1 = [pad + 'lambda ' + pos(e) + ' params: [' + e.params.join(', ') + ']'];
        l1.push(renderAst(e.body, depth + 1));
        return l1.join('\n');
      }
      case 'apply': {
        var l2 = [pad + 'apply ' + pos(e)];
        l2.push(renderAst(e.fn, depth + 1));
        l2.push(renderAst(e.arg, depth + 1));
        return l2.join('\n');
      }
      case 'let': {
        var l3 = [pad + 'let' + (e.rec ? ' rec' : '') + ' ' + e.name + ' ' + pos(e)];
        l3.push(pad + '  bound:');
        l3.push(renderAst(e.bound, depth + 2));
        l3.push(pad + '  body:');
        l3.push(renderAst(e.body, depth + 2));
        return l3.join('\n');
      }
      case 'if': {
        var l4 = [pad + 'if ' + pos(e)];
        l4.push(pad + '  cond:');
        l4.push(renderAst(e.cond, depth + 2));
        l4.push(pad + '  then:');
        l4.push(renderAst(e.then, depth + 2));
        l4.push(pad + '  else:');
        l4.push(renderAst(e.els, depth + 2));
        return l4.join('\n');
      }
      case 'match': {
        var lm = [pad + 'match ' + pos(e)];
        lm.push(pad + '  scrut:');
        lm.push(renderAst(e.scrut, depth + 2));
        e.branches.forEach(function (b) {
          lm.push(pad + '  branch:');
          lm.push(pad + '    pattern:');
          lm.push(renderPattern(b[0], depth + 3));
          lm.push(pad + '    body:');
          lm.push(renderAst(b[1], depth + 3));
        });
        return lm.join('\n');
      }
      case 'bin': {
        var l5 = [pad + 'bin ' + e.op + ' ' + pos(e)];
        l5.push(renderAst(e.a, depth + 1));
        l5.push(renderAst(e.b, depth + 1));
        return l5.join('\n');
      }
      case 'neg': {
        var l6 = [pad + 'neg ' + pos(e)];
        l6.push(renderAst(e.x, depth + 1));
        return l6.join('\n');
      }
      case 'not': {
        var l7 = [pad + 'not ' + pos(e)];
        l7.push(renderAst(e.x, depth + 1));
        return l7.join('\n');
      }
      case 'record': {
        var lr = [pad + 'record ' + pos(e)];
        e.fields.forEach(function (f) {
          lr.push(pad + '  ' + f[0] + ' =');
          lr.push(renderAst(f[1], depth + 2));
        });
        return lr.join('\n');
      }
      case 'proj': {
        var lpr = [pad + 'proj ' + e.name + ' ' + pos(e)];
        lpr.push(renderAst(e.e, depth + 1));
        return lpr.join('\n');
      }
      case 'update': {
        var lu = [pad + 'update ' + e.name + ' ' + pos(e)];
        lu.push(pad + '  base:');
        lu.push(renderAst(e.e, depth + 2));
        lu.push(pad + '  value:');
        lu.push(renderAst(e.value, depth + 2));
        return lu.join('\n');
      }
      default: return pad + '?';
    }
  }

  function renderPattern(pat, depth) {
    var pad = new Array(depth * 2 + 1).join(' ');
    function pos(p) { return p.pos ? '@' + p.pos.line + ':' + p.pos.col : ''; }
    switch (pat.kind) {
      case 'pint':   return pad + 'pint ' + pat.v + ' ' + pos(pat);
      case 'pfloat': return pad + 'pfloat ' + pat.v + ' ' + pos(pat);
      case 'pbool':  return pad + 'pbool ' + pat.v + ' ' + pos(pat);
      case 'pstr':   return pad + 'pstr "' + pat.v + '" ' + pos(pat);
      case 'pchar':  return pad + 'pchar ' + pat.v + ' ' + pos(pat);
      case 'pvar':   return pad + 'pvar ' + pat.name + ' ' + pos(pat);
      case 'pwild':  return pad + 'pwild ' + pos(pat);
      case 'pnil':   return pad + 'pnil ' + pos(pat);
      case 'precord': {
        var pr = [pad + 'precord ' + pos(pat)];
        pat.fields.forEach(function (f) {
          pr.push(pad + '  ' + f[0] + ' =');
          pr.push(renderPattern(f[1], depth + 2));
        });
        return pr.join('\n');
      }
      case 'pcons': {
        var pc = [pad + 'pcons ' + pos(pat)];
        pc.push(renderPattern(pat.h, depth + 1));
        pc.push(renderPattern(pat.t, depth + 1));
        return pc.join('\n');
      }
      case 'plist': {
        var pl = [pad + 'plist ' + pos(pat)];
        pat.items.forEach(function (p) { pl.push(renderPattern(p, depth + 1)); });
        return pl.join('\n');
      }
      case 'pconstr': {
        var pk = [pad + 'pconstr ' + pat.name + ' ' + pos(pat)];
        pat.args.forEach(function (p) { pk.push(renderPattern(p, depth + 1)); });
        return pk.join('\n');
      }
      default: return pad + '?';
    }
  }

  function renderProgram(prog) {
    var lines = [];
    (prog.imports || []).forEach(function (imp) {
      lines.push('import "' + imp + '"');
    });
    (prog.defs || []).forEach(function (d) {
      lines.push(renderDef(d));
    });
    if (prog.expr !== null && prog.expr !== undefined) {
      lines.push(renderAst(prog.expr, 0));
    }
    return lines.join('\n');
  }

  function renderDef(d) {
    switch (d.kind) {
      case 'defdata': {
        var dd = d.decl;
        var ctorParts = dd.ctors.map(function (c) {
          return c.name + (c.fields.length ? ' ' + c.fields.map(renderType).join(' ') : '');
        });
        return 'data ' + dd.name + (dd.tyvars.length ? ' ' + dd.tyvars.join(' ') : '')
          + ' = ' + ctorParts.join(' | ');
      }
      case 'defrecord': {
        var dr = d.decl;
        return 'record ' + dr.name + (dr.tyvars.length ? ' ' + dr.tyvars.join(' ') : '')
          + ' = { ' + dr.fields.map(function (f) { return f[0] + ' : ' + renderType(f[1]); }).join(', ') + ' }';
      }
      case 'defclass': {
        var dc = d.decl;
        return 'class ' + dc.name + ' ' + dc.var + ' where\n'
          + dc.methods.map(function (m) { return '  ' + m[0] + ' : ' + renderType(m[1]); }).join('\n');
      }
      case 'definstance': {
        var di = d.decl;
        var ctx = '';
        if (di.ctx) {
          ctx = ' ' + di.ctx[0] + ' ' + di.ctx[1] + ' =>';
        }
        return 'instance' + ctx + ' ' + di.class + ' ' + renderType(di.head) + ' where\n'
          + di.methods.map(function (m) { return '  ' + m[0] + ' = ' + renderAst(m[1], 0); }).join('\n');
      }
      case 'deflet':
        return 'let' + (d.rec ? ' rec' : '') + ' ' + d.name + ' = ' + renderAst(d.bound, 0);
      case 'definfix':
        return (d.assoc === 'left' ? 'infixl ' : (d.assoc === 'right' ? 'infixr ' : 'infix '))
          + d.level + ' ' + d.name;
      case 'defsynonym':
        return 'type ' + d.name + (d.tyvars.length ? ' ' + d.tyvars.join(' ') : '')
          + ' = ' + renderType(d.ty);
      default:
        return '?';
    }
  }

  function renderType(t) {
    switch (t.k) {
      case 'int': return 'Int';
      case 'float': return 'Float';
      case 'bool': return 'Bool';
      case 'str': return 'String';
      case 'char': return 'Char';
      case 'unit': return '()';
      case 'effect': return 'Effect ' + renderType(t.t);
      case 'var': return t.n >= 2000000000 ? 'a' : 'a' + t.n;
      case 'list': return '[' + renderType(t.t) + ']';
      case 'fun': return renderType(t.a) + ' -> ' + renderType(t.b);
      case 'data': return t.n + (t.args.length ? ' ' + t.args.map(renderType).join(' ') : '');
      case 'rec': return t.n + (t.args.length ? ' ' + t.args.map(renderType).join(' ') : '');
      default: return '?';
    }
  }

  // -------------------------------------------------------------------
  // Actions
  // -------------------------------------------------------------------
  document.getElementById('run-btn').addEventListener('click', function () {
    var src = source.value;
    var ins = stdinLines();
    var inf = H.inferProgram(src);
    if (!inf.ok) { showError('output', inf); activate('output'); return; }
    var r = H.evalProgramEffect(ins, src);
    if (r.kind === 'eval') { showError('output', r); activate('output'); return; }
    var v = r.value;
    var unitK = v && (v.k === 'unit' || v.k === 'vm_unit');
    show('output', r.out + (unitK ? '' : H.showValue(v)));
    activate('output');
  });

  document.getElementById('run-vm-btn').addEventListener('click', function () {
    var src = source.value;
    var ins = stdinLines();
    var inf = H.inferProgram(src);
    if (!inf.ok) { showError('output', inf); activate('output'); return; }
    var cp = H.compileProgram(src, document.getElementById('opt-check').checked);
    if (cp.kind) { showError('output', cp); activate('output'); return; }
    var res = H.runVmEffect(cp.program, ins);
    if (res.kind) { showError('output', res); activate('output'); return; }
    var v = res.value;
    var unitK = v && (v.k === 'unit' || v.k === 'vm_unit');
    show('output', res.out + (unitK ? '' : H.vmShowValue(v)));
    activate('output');
  });

  document.getElementById('type-btn').addEventListener('click', function () {
    var inf = H.inferProgram(source.value);
    if (!inf.ok) { showError('type', inf); activate('type'); return; }
    show('type', H.showType(inf.type));
    activate('type');
  });

  document.getElementById('profile-btn').addEventListener('click', function () {
    var inf = H.inferProgram(source.value);
    if (!inf.ok) { showError('profile', inf); activate('profile'); return; }
    var cp = H.compileProgram(source.value, document.getElementById('opt-check').checked);
    if (cp.kind) { showError('profile', cp); activate('profile'); return; }
    var res = H.runVmProfiled(cp.program);
    if (res.kind) { showError('profile', res); activate('profile'); return; }
    show('profile', H.renderProfile(res.profile) + '\n\nvalue: ' + H.vmShowValue(res.value));
    activate('profile');
  });

  document.getElementById('ast-btn').addEventListener('click', function () {
    var p = H.parse(source.value);
    if (p.kind) { showError('ast', p); activate('ast'); return; }
    show('ast', renderProgram(p));
    activate('ast');
  });

  document.getElementById('bytecode-btn').addEventListener('click', function () {
    var cp = H.compileProgram(source.value, document.getElementById('opt-check').checked);
    if (cp.kind) { showError('bytecode', cp); activate('bytecode'); return; }
    show('bytecode', H.disassemble(cp.program.entry));
    activate('bytecode');
  });

  // Serialize the compiled program to a .hbc artifact (same format the CLI
  // emits with `compile -o`); the Download button saves it as a file.
  var artifactText = null;
  function buildArtifact() {
    var cp = H.compileProgram(source.value, document.getElementById('opt-check').checked);
    if (cp.kind) { showError('artifact', cp); activate('artifact'); return false; }
    var ser = H.serializeProgram(cp.program);
    if (!ser.ok) { showError('artifact', ser); activate('artifact'); return false; }
    artifactText = ser.text;
    show('artifact', ser.text);
    activate('artifact');
    return true;
  }
  var artifactTab = document.querySelector('.tab[data-panel="artifact"]');
  artifactTab.addEventListener('click', buildArtifact);
  document.getElementById('artifact-dl').addEventListener('click', function () {
    if (artifactText === null) {
      if (!buildArtifact()) { return; }
    }
    var blob = new Blob([artifactText], { type: 'text/plain' });
    var a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'program.hbc';
    a.click();
    URL.revokeObjectURL(a.href);
  });

  // -------------------------------------------------------------------
  // Step-through VM debugger
  // -------------------------------------------------------------------
  var stepper = null;
  var dbgInstr = document.getElementById('dbg-instr');
  var dbgFrame = document.getElementById('dbg-frame');
  var dbgResult = document.getElementById('dbg-result');
  var dbgStack = document.getElementById('dbg-stack');
  var stepBtn = document.getElementById('dbg-step');
  var runBtn = document.getElementById('dbg-run');

  function buildStepper() {
    var inf = H.inferProgram(source.value);
    if (!inf.ok) {
      dbgInstr.innerHTML = 'current instruction: <span class="error">type error: ' + inf.message + '</span>';
      return null;
    }
    var cp = H.compileProgram(source.value);
    if (cp.kind) {
      dbgInstr.innerHTML = 'current instruction: <span class="error">compile error: ' + cp.message + '</span>';
      return null;
    }
    return H.makeStepper(cp.program);
  }

  function renderDebugger() {
    if (!stepper) {
      dbgStack.textContent = 'stack (top first): empty';
      dbgFrame.textContent = 'frame depth: 0';
      dbgResult.textContent = 'result: -';
      stepBtn.disabled = true;
      runBtn.disabled = true;
      return;
    }
    var s = stepper.snapshot();
    stepBtn.disabled = s.done;
    runBtn.disabled = s.done;
    if (s.done) {
      if (s.result && s.result.kind === 'vm') {
        dbgInstr.innerHTML = 'current instruction: <span class="error">vm error: ' + s.result.message + '</span>';
        dbgResult.innerHTML = 'result: <span class="error">' + s.result.message + '</span>';
      } else if (s.result) {
        dbgInstr.innerHTML = 'current instruction: <span class="muted">halt</span>';
        dbgResult.innerHTML = 'result: <strong>' + H.vmShowValue(s.result.value) + '</strong>';
      }
    } else {
      var instr = s.instr ? H.showInstr(s.instr) : '';
      dbgInstr.innerHTML = 'current instruction: <strong>' + instr + '</strong>'
        + ' <span class="muted">(' + s.fnName + ' @' + s.ip + ')</span>';
      dbgResult.textContent = 'result: -';
    }
    dbgFrame.textContent = 'frame depth: ' + s.frameDepth;
    if (s.stack.length === 0) {
      dbgStack.innerHTML = 'stack (top first): <span class="muted">empty</span>';
    } else {
      var parts = s.stack.map(function (v) {
        return '<span class="' + (v === s.stack[s.stack.length - 1] ? 'stack-top' : '') + '">' + H.vmShowValue(v) + '</span>';
      });
      dbgStack.innerHTML = 'stack (top first): ' + parts.join(' | ');
    }
  }

  document.getElementById('dbg-reset').addEventListener('click', function () {
    stepper = buildStepper();
    renderDebugger();
  });
  stepBtn.addEventListener('click', function () {
    if (!stepper) { stepper = buildStepper(); }
    stepper.step();
    renderDebugger();
  });
  runBtn.addEventListener('click', function () {
    if (!stepper) { stepper = buildStepper(); }
    for (var i = 0; i < 100000000; i++) {
      var s = stepper.snapshot();
      if (s.done) { break; }
      stepper.step();
    }
    renderDebugger();
  });

  renderDebugger();
})();
